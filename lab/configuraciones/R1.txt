! ============================================================
! R1 - Cisco ISR4331 (Packet Tracer 9.0.1)
! LAN1 192.168.10.0/24 + enlace WAN 10.0.12.0/30 hacia R2
! Pegar en modo EXEC privilegiado (Router#) o linea por linea.
! ============================================================
enable
configure terminal
hostname R1
no ip domain-lookup
enable secret class
banner motd #Laboratorio CE-5301 - Acceso solo autorizado#
!
interface GigabitEthernet0/0/0
 description Enlace WAN hacia R2
 ip address 10.0.12.1 255.255.255.252
 no shutdown
 exit
interface GigabitEthernet0/0/1
 description LAN1 - red de usuarios
 ip address 192.168.10.1 255.255.255.0
 no shutdown
 exit
interface GigabitEthernet0/0/2
 description Sin uso
 shutdown
 exit
interface Loopback0
 description Identificador del router (router-id OSPF)
 ip address 1.1.1.1 255.255.255.255
 exit
!
router ospf 1
 router-id 1.1.1.1
 network 10.0.12.0 0.0.0.3 area 0
 network 192.168.10.0 0.0.0.255 area 0
 network 1.1.1.1 0.0.0.0 area 0
 passive-interface GigabitEthernet0/0/1
 exit
!
line console 0
 logging synchronous
 exit
end
copy running-config startup-config
